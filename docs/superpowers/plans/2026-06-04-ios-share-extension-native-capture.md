# iOS Share Extension Native Capture Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make Mentioned appear in the iOS share sheet for shared URL/text payloads, receive a supported Instagram Reel/post URL, and hand that URL to the host app without creating a backend job.

**Architecture:** Keep this as a managed Expo/CNG change. Use `expo-share-extension` to generate the native iOS share-extension target during prebuild/EAS, keep extraction/job creation in the main app/backend, and hand off shared payloads through `mentioned://share?url=...`. The host app only stages the shared URL in the existing paste flow; issue 14 owns automatic job creation.

**Tech Stack:** Expo SDK 54, React Native 0.81, TypeScript, `expo-share-extension`, Expo Linking, EAS iOS app extension metadata, XcodeBuildMCP validation after prebuild.

- [ ] **Step 5: Implement the URL utility**

Create `mobile/src/utils/shared-source-url.ts`:

```ts
export type SharedSourcePayload = {
  url?: string;
  text?: string;
};

const URL_PATTERN = /https?:\/\/[^\s<>"']+/gi;
const TRAILING_PUNCTUATION_PATTERN = /[),.;!?]+$/;
const INSTAGRAM_HOSTS = new Set(['instagram.com', 'www.instagram.com']);
const TRACKING_PARAMS = new Set([
  'igsh',
  'utm_campaign',
  'utm_content',
  'utm_medium',
  'utm_source',
  'utm_term',
]);

export function isSupportedSharedSourceUrl(value: string): boolean {
  const parsed = parseUrl(value);
  if (!parsed) {
    return false;
  }

  return (
    parsed.protocol === 'https:' &&
    INSTAGRAM_HOSTS.has(parsed.hostname) &&
    (parsed.pathname.toLowerCase().includes('/reel/') || parsed.pathname.toLowerCase().includes('/p/'))
  );
}

export function extractSharedSourceUrl(payload: SharedSourcePayload): string | null {
  const candidates = candidateUrls(payload);

  for (const candidate of candidates) {
    const normalized = normalizeSharedSourceUrl(candidate);
    if (normalized && isSupportedSharedSourceUrl(normalized)) {
      return normalized;
    }
  }

  return null;
}

export function sharedUrlFromMentionedDeepLink(rawUrl: string): string | null {
  const parsed = parseUrl(rawUrl);
  if (!parsed) {
    return null;
  }

  const route = parsed.hostname || parsed.pathname.replace(/^\/+/, '');
  if (parsed.protocol !== 'mentioned:' || route !== 'share') {
    return null;
  }

  const sharedUrl = parsed.searchParams.get('url');
  if (!sharedUrl) {
    return null;
  }

  return extractSharedSourceUrl({ url: sharedUrl });
}

function candidateUrls(payload: SharedSourcePayload): string[] {
  const candidates: string[] = [];

  if (payload.url) {
    candidates.push(payload.url);
  }

  if (payload.text) {
    candidates.push(...Array.from(payload.text.matchAll(URL_PATTERN), (match) => match[0]));
  }

  return candidates;
}

function normalizeSharedSourceUrl(value: string): string | null {
  const stripped = value.trim().replace(TRAILING_PUNCTUATION_PATTERN, '');
  const parsed = parseUrl(stripped);
  if (!parsed) {
    return null;
  }

  for (const key of Array.from(parsed.searchParams.keys())) {
    if (TRACKING_PARAMS.has(key.toLowerCase())) {
      parsed.searchParams.delete(key);
    }
  }

  parsed.hash = '';
  return parsed.toString();
}

function parseUrl(value: string): URL | null {
  try {
    return new URL(value.trim());
  } catch {
    return null;
  }
}
```

import {
  close,
  openHostApp,
  type InitialProps,
  Text,
  View,
} from 'expo-share-extension';
import { Pressable, StyleSheet } from 'react-native';

import { extractSharedSourceUrl } from './src/utils/shared-source-url';

export default function ShareExtension(props: InitialProps) {
  const sourceUrl = extractSharedSourceUrl({ url: props.url, text: props.text });

  const openMentioned = () => {
    if (!sourceUrl) {
      close();
      return;
    }

    openHostApp(`share?url=${sourceUrl}`);
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>{sourceUrl ? 'Save to Mentioned' : 'Unsupported source'}</Text>
      <Text style={styles.body}>
        {sourceUrl ? 'Open Mentioned to save this source.' : 'Share an Instagram Reel or post link.'}
      </Text>
      {sourceUrl ? (
        <Text ellipsizeMode="middle" numberOfLines={2} style={styles.url}>
          {sourceUrl}
        </Text>
      ) : null}
      <View style={styles.actions}>
        {sourceUrl ? (
